  \def\FrameCommand{{\color{#1}\vrule width 3pt}\hspace{8pt}}%
  \MakeFramed{\advance\hsize-\width\FrameRestore}\small}%
 {\endMakeFramed}

\newenvironment{aviso}{\begin{caixaaurora}{statusWarning}}{\end{caixaaurora}}
\newenvironment{perigo}{\begin{caixaaurora}{statusError}}{\end{caixaaurora}}
\newenvironment{info}{\begin{caixaaurora}{auroraCyan}}{\end{caixaaurora}}
